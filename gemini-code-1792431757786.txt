    <!-- All Scripting Logic -->
    <script>
        document.getElementById('year').textContent = new Date().getFullYear();

        // -----------------------------------------------------
        // 1. Navigation & Menu Setup
        // -----------------------------------------------------
        const btn = document.getElementById('mobile-menu-btn');
        const menu = document.getElementById('mobile-menu');
        const icon = btn.querySelector('i');

        btn.addEventListener('click', () => {
            menu.classList.toggle('open');
            if (menu.classList.contains('open')) {
                icon.classList.remove('fa-bars');
                icon.classList.add('fa-xmark');
            } else {
                icon.classList.remove('fa-xmark');
                icon.classList.add('fa-bars');
            }
        });

        function navigateTo(pageId) {
            const sections = document.querySelectorAll('.page-section');
            sections.forEach(sec => sec.classList.remove('active'));

            const target = document.getElementById(pageId);
            if (target) target.classList.add('active');

            const navLinks = document.querySelectorAll('.nav-link');
            navLinks.forEach(link => {
                link.classList.remove('active', 'text-brandBlue-500');
                link.classList.add('text-gray-600');
                if (link.getAttribute('href') === '#' + pageId) {
                    link.classList.add('active', 'text-brandBlue-500');
                    link.classList.remove('text-gray-600');
                }
            });

            if (menu.classList.contains('open')) {
                menu.classList.remove('open');
                icon.classList.remove('fa-xmark');
                icon.classList.add('fa-bars');
            }

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function showMessage(msg, alertId, formSelector) {
            const alertBox = document.getElementById(alertId);
            if(alertBox) {
                alertBox.textContent = msg;
                alertBox.classList.remove('hidden');
                setTimeout(() => alertBox.classList.add('hidden'), 3000);
            }
            if(formSelector) {
                const formElement = document.querySelector(formSelector);
                if(formElement) formElement.reset();
            }
        }

        // -----------------------------------------------------
        // 2. Future-Proof Job Posters Array Configuration
        // -----------------------------------------------------
        /* 
           To add a job poster permanently in the future, simply add a new line 
           to this array following the format: { title: "Job Name", img: "filename.png" } 
        */
        const initialJobPosters = [
            { title: "Software Engineer", img: "job-poster-1.png" },
            { title: "HR Executive", img: "job-poster-2.png" },
            { title: "Marketing Manager", img: "job-poster-3.png" }
        ];

        // -----------------------------------------------------
        // 3. Render Job Posters Function
        // -----------------------------------------------------
        function renderJobPosters() {
            const grid = document.getElementById('job-posters-grid');
            if (!grid) return;
            
            grid.innerHTML = ''; // Clear current posters
            
            initialJobPosters.forEach((poster) => {
                const posterCard = `
                    <div class="bg-white rounded-2xl overflow-hidden shadow border border-gray-100 hover:shadow-2xl hover:-translate-y-2 transition-all duration-300 group flex flex-col relative">
                        <!-- Poster Image area -->
                        <div class="relative h-96 w-full bg-gray-100 overflow-hidden">
                            <img src="${poster.img}" alt="${poster.title}" class="w-full h-full object-cover transform group-hover:scale-110 transition-transform duration-700" onerror="this.src='https://images.unsplash.com/photo-1586281380349-632531db7ed4?auto=format&fit=crop&w=600&q=80'">
                            
                            <!-- Hover Overlay for Apply Button -->
                            <div class="absolute inset-0 bg-gradient-to-t from-navy-900/90 via-navy-900/40 to-transparent opacity-0 group-hover:opacity-100 transition-opacity duration-300 flex items-center justify-center">
                                <button onclick="navigateTo('register')" class="px-8 py-3 bg-brandBlue-500 text-white rounded-full font-bold shadow-lg hover:bg-brandBlue-600 transform translate-y-8 group-hover:translate-y-0 transition-all duration-300 delay-75">
                                    Apply Now <i class="fa-solid fa-arrow-right ml-2"></i>
                                </button>
                            </div>
                        </div>
                        <!-- Bottom details -->
                        <div class="p-5 text-center border-t border-gray-100 bg-white z-10">
                            <h3 class="text-xl font-bold text-navy-900">${poster.title}</h3>
                        </div>
                    </div>
                `;
                grid.insertAdjacentHTML('beforeend', posterCard);
            });
        }

        // -----------------------------------------------------
        // 4. Handle "Add New Poster" Modal Interface
        // -----------------------------------------------------
        function toggleAddPosterModal() {
            const modal = document.getElementById('add-poster-modal');
            modal.classList.toggle('hidden');
        }

        function handleAddNewPoster(event) {
            event.preventDefault();
            
            // Get user input from the form
            const titleInput = document.getElementById('poster-title').value;
            const imgInput = document.getElementById('poster-img').value;
            
            // Add the new poster to the beginning of our array
            initialJobPosters.unshift({ 
                title: titleInput, 
                img: imgInput 
            });
            
            // Instantly re-render the grid to show the new addition
            renderJobPosters();
            
            // Close the modal and reset the form
            toggleAddPosterModal();
            event.target.reset();
            
            alert(`Success! "${titleInput}" has been added to the Job Board.`);
        }

        // -----------------------------------------------------
        // 5. Initialize the Page on Load
        // -----------------------------------------------------
        window.onload = () => {
            // Render default images
            const galleryGrid = document.getElementById('gallery-grid');
            if (galleryGrid) {
                let galleryHTML = '';
                for (let i = 1; i <= 54; i++) {
                    galleryHTML += `
                        <div class="break-inside-avoid rounded-2xl overflow-hidden group relative shadow-sm hover:shadow-xl transition-all duration-300 cursor-pointer">
                            <img src="gallery (${i}).png" loading="lazy" alt="Gallery Image ${i}" class="w-full h-auto object-cover transform group-hover:scale-110 transition-transform duration-700 bg-gray-200" onerror="this.src='https://via.placeholder.com/600x400?text=Gallery+Image+${i}'">
                        </div>
                    `;
                }
                galleryGrid.innerHTML = galleryHTML;
            }

            // Render Jobs
            renderJobPosters();

            // Handle routing
            const hash = window.location.hash.replace('#', '');
            if (hash && document.getElementById(hash)) {
                navigateTo(hash);
            } else {
                navigateTo('home');
            }
        };
    </script>
